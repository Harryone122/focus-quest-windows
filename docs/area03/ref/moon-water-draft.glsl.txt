// 夜晚預設改成：sun: 0xc4d3ec, sunI: 0.62, sunAt: [9, 17, 13], hS: 0x5d7098, hG: 0x18202f, hI: 0.78（數值是起點，看畫面調）。

// the reflected full moon (Opus 2026-10-01). Calm water (high tide) gives a clean disc; wind ripples (low tide) cut it into slices
// and scatter a road of broken light from the disc toward the viewer. uMoon = (u, v, radius); v is divided by uMoonK so it reads round.
'  float moonNight = smoothstep(0.65, 0.97, uStar);',
'  float calmK = clamp(uRipple, 0.0, 1.0) * ringMask;',
'  vec2 mq = uv - uMoon.xy; mq.y /= uMoonK;',
'  float mjit = (vn(vec2(mq.y * 7.0 / uMoon.z + uTime * 0.45, uTime * 0.21)) - 0.5) * 0.5 * calmK;',
'  vec2 mqd = vec2(mq.x + mjit * uMoon.z * 0.45, mq.y);',
'  float mr = length(mqd) / uMoon.z;',
'  float slice = 1.0 - calmK * 0.6 * smoothstep(0.5, 0.82, vn(vec2(mq.y * 12.0 / uMoon.z, uTime * 0.8)));',
'  float moonDisk = smoothstep(1.0, 0.95, mr);',
'  vec2 mn = mqd / uMoon.z;',
'  float maria = 0.94 + 0.06 * fbm(mn * 2.2 + vec2(3.1, 7.7)) - 0.07 * exp(-pow(length(mn - vec2(-0.28, 0.2)) * 3.0, 2.0)) - 0.05 * exp(-pow(length(mn - vec2(0.24, -0.18)) * 3.6, 2.0)) - 0.035 * exp(-pow(length(mn - vec2(0.05, 0.42)) * 4.5, 2.0));',
'  vec3 moonC = vec3(0.99, 0.97, 0.90) * maria * (1.0 - 0.10 * mr * mr);',
'  float glow = exp(-max(mr - 1.0, 0.0) * 2.6) * (1.0 - moonDisk);',
'  float halo = exp(-mr * 0.40);',
'  float gl = -mq.y / uMoon.z;',
'  float gw = 0.9 + 0.07 * max(gl, 0.0);',
'  float across = mq.x / uMoon.z / gw;',
'  float road = exp(-across * across * 1.5) * smoothstep(-0.3, 1.5, gl) * (1.0 - smoothstep(5.0, 15.0, gl));',
'  float dash = smoothstep(0.6, 0.86, vn(vec2(uv.x * 2.6 + uTime * 0.13, uv.y * 8.0 - uTime * 0.6)));',
'  dash *= 0.35 + 0.65 * smoothstep(0.3, 0.7, vn(vec2(uv.x * 0.9 - uTime * 0.07, uv.y * 2.4 + uTime * 0.11)));',
'  vec3 moonSheen = vec3(0.62, 0.71, 0.86);',
'  sky += uStar * star * vec3(1.0, 0.97, 0.9) * (1.0 - cov * 0.85) * (1.0 - 0.8 * halo * moonNight);',
'  sky = mix(sky, uHaze, hazeK * 0.9);',
'  sky += moonSheen * (halo * 0.09 + glow * 0.30) * moonNight;',
'  sky += moonSheen * road * (0.04 + 0.09 * calmK) * moonNight;',
'  sky += vec3(0.96, 0.94, 0.86) * road * dash * (0.95 * calmK + 0.06) * moonNight;',
'  sky = mix(sky, moonC, moonDisk * slice * moonNight);',
